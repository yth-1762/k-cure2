data5$grouptype <- relevel(data5$grouptype, ref="002")
data5$non_small_yn <- relevel(data5$non_small_yn,ref="1")
m1<-model.matrix(~grouptype+SEX+G1E_BMI+income
                 +cci_group+activity+non_small_yn+SEER_GRP+surgery_yn+radio_yn+system_yn,data=data5)[,-1] 
View(m1)
crr5_2 <- crr(ftime=data5$day_diff,fstatus=data5$dth_type, cencode=0, cov1=m1, failcode=1)
summary(crr5_2)
exp(crr5_2$coef)
beta<- crr5_2$coef
se <- sqrt(diag(crr5_2$var))
hr <- exp(beta)
lower <- exp(beta - 1.96*se)
upper <- exp(beta + 1.96*se)
result <- data.frame(HR = hr, lower95=lower, upper95=upper)
result
s<- summary(crr5_2)
zval <- s$coef[,"z"]
pval <- 2*(1-pnorm(abs(zval)))
cbind(s$coef,pval)
